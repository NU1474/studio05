        function createRackMesh(w, h, d) {
            const group = new THREE.Group();

            // Metallic Frame
            const frameMat = new THREE.MeshStandardMaterial({ color: 0x1e293b, roughness: 0.5, metalness: 0.8 });
            const thickness = 0.03;

            // Frame vertical pillars
            const pGeo = new THREE.BoxGeometry(thickness, h, thickness);
            const p1 = new THREE.Mesh(pGeo, frameMat); p1.position.set(-w/2 + thickness/2, 0, -d/2 + thickness/2);
            const p2 = new THREE.Mesh(pGeo, frameMat); p2.position.set(w/2 - thickness/2, 0, -d/2 + thickness/2);
            const p3 = new THREE.Mesh(pGeo, frameMat); p3.position.set(-w/2 + thickness/2, 0, d/2 - thickness/2);
            const p4 = new THREE.Mesh(pGeo, frameMat); p4.position.set(w/2 - thickness/2, 0, d/2 - thickness/2);
            group.add(p1, p2, p3, p4);

            // Battery Packs Rendering
            const packCount = parseInt(state.battery.packsPerRack) || 8;
            const packMat = new THREE.MeshStandardMaterial({
                color: state.colors.battery,
                roughness: 0.4,
                metalness: 0.3
            });

            const packH = (h - 0.2) / packCount;
            const packGeo = new THREE.BoxGeometry(w - 0.08, packH * 0.8, d - 0.08);

            for (let i = 0; i < packCount; i++) {
                const packMesh = new THREE.Mesh(packGeo, packMat);
                const yPos = -h/2 + 0.1 + (i * packH) + (packH / 2);
                packMesh.position.set(0, yPos, 0);
                packMesh.castShadow = true;
                group.add(packMesh);

                // Individual Cell detailing indicators on pack front face
                const cellIndicatorMat = new THREE.MeshBasicMaterial({ color: 0x38bdf8 });
                const cellIndicatorGeo = new THREE.BoxGeometry((w - 0.12), 0.015, 0.01);
                const cellInd = new THREE.Mesh(cellIndicatorGeo, cellIndicatorMat);
                cellInd.position.set(0, yPos, d/2 - 0.03);
                group.add(cellInd);
            }

            return group;
        }

        function createEquipmentMesh(w, h, d, colorHex, labelName) {
            const group = new THREE.Group();
            const mainGeo = new THREE.BoxGeometry(w, h, d);
            const mainMat = new THREE.MeshStandardMaterial({
                color: colorHex,
                roughness: 0.5,
                metalness: 0.4

            const mainMesh = new THREE.Mesh(mainGeo, mainMat);

        function drawWalkwayDimensions() {
            while (dimensionGroup.children.length > 0) {
                dimensionGroup.remove(dimensionGroup.children[0]);
            }

            const spec = CONTAINER_SPECS[state.containerSize];
            const cW = spec.width / 1000;
            const cL = spec.length / 1000;

            // Compute central maintenance aisle width
            let minPositiveZ = cW / 2;
            let maxNegativeZ = -cW / 2;

            interactiveObjects.forEach(obj => {
                const z = obj.position.z;
                if (z > 0 && z < minPositiveZ) minPositiveZ = z - 0.4;
                if (z < 0 && z > maxNegativeZ) maxNegativeZ = z + 0.4;
            });

            const walkwayWidthMM = Math.max(0, Math.round((minPositiveZ - maxNegativeZ) * 1000));
            document.getElementById('valWalkway').innerText = `${walkwayWidthMM} mm`;

            // Draw Dashed Line in Scene
            const points = [];
            points.push(new THREE.Vector3(-cL / 3, 0.05, maxNegativeZ));
            points.push(new THREE.Vector3(-cL / 3, 0.05, minPositiveZ));

            const geom = new THREE.BufferGeometry().setFromPoints(points);
            const mat = new THREE.LineDashedMaterial({
                color: 0xc084fc,
                dashSize: 0.2,
                gapSize: 0.1,
                linewidth: 2
            });
            const line = new THREE.Line(geom, mat);
            line.computeLineDistances();
            dimensionGroup.add(line);
        }

        function setupInteractions() {
            const dom = renderer.domElement;

            dom.addEventListener('pointerdown', (e) => {
                const rect = dom.getBoundingClientRect();
                mouse.x = ((e.clientX - rect.left) / dom.clientWidth) * 2 - 1;
                mouse.y = -((e.clientY - rect.top) / dom.clientHeight) * 2 + 1;

                raycaster.setFromCamera(mouse, camera);
                const intersects = raycaster.intersectObjects(interactiveObjects, true);

                if (intersects.length > 0) {
                    let rootObj = intersects[0].object;
                    while (rootObj.parent && rootObj.parent !== scene) {
                        rootObj = rootObj.parent;
                    }
                    selectObject(rootObj);

                    isDragging = true;
                    controls.enabled = false;

                    const intersectionPoint = new THREE.Vector3();
                    raycaster.ray.intersectPlane(plane, intersectionPoint);
                    dragOffset.subVectors(rootObj.position, intersectionPoint);
                }
            });

            dom.addEventListener('pointermove', (e) => {
                if (!isDragging || !selectedObject) return;

                const rect = dom.getBoundingClientRect();
                mouse.x = ((e.clientX - rect.left) / dom.clientWidth) * 2 - 1;
                mouse.y = -((e.clientY - rect.top) / dom.clientHeight) * 2 + 1;

                raycaster.setFromCamera(mouse, camera);
                const intersectionPoint = new THREE.Vector3();
                if (raycaster.ray.intersectPlane(plane, intersectionPoint)) {
                    const newPos = intersectionPoint.add(dragOffset);
                    
                    // Container boundaries constraint
                    const spec = CONTAINER_SPECS[state.containerSize];
                    const maxL = spec.length / 2000 - 0.3;
                    const maxW = spec.width / 2000 - 0.3;

                    selectedObject.position.x = Math.max(-maxL, Math.min(maxL, newPos.x));
                    selectedObject.position.z = Math.max(-maxW, Math.min(maxW, newPos.z));

                    // Save state position
                    const id = selectedObject.userData.id;
                    if (id) {
                        if (!state.placements[id]) state.placements[id] = {};
                        state.placements[id].pos = selectedObject.position.clone();
                    }

                    drawWalkwayDimensions();
                    updateCalculations();
                    autoSave();
                }
            });

            const stopDrag = () => {
                isDragging = false;
                controls.enabled = true;
            };

            dom.addEventListener('pointerup', stopDrag);
            dom.addEventListener('pointerleave', stopDrag);

        function updateCalculations() {
            // 1. Capacity Calculation
            // Cell voltage = 3.2V (LFP)
            const cellCapAh = parseFloat(state.battery.capacityAh) || 306;
            const cellsPerPack = parseInt(state.battery.cellsPerPack) || 16;
            const packsPerRack = parseInt(state.battery.packsPerRack) || 8;
            const totalRacks = (parseInt(state.battery.rackCols) || 2) * (parseInt(state.battery.rackRows) || 4);

            const packWh = cellCapAh * 3.2 * cellsPerPack; // Wh
            const rackKWh = (packWh * packsPerRack) / 1000; // kWh
            const totalMWh = (rackKWh * totalRacks) / 1000; // MWh

            document.getElementById('valTotalCap').innerText = `${totalMWh.toFixed(3)} MWh`;
            document.getElementById('valSubCap').innerText = `${totalRacks} Racks / ${(rackKWh * totalRacks).toFixed(1)} kWh`;

            // 2. Weight Calculations
            const packWeightKg = parseFloat(state.battery.packWeight) || 110;
            const totalBatteryWeightKg = packWeightKg * packsPerRack * totalRacks;

            const pcsKg = parseFloat(state.pcs.weight) || 0;
            const invKg = parseFloat(state.inv.weight) || 0;
            const hvacKg = parseFloat(state.hvac.weight) || 0;
            const novKg = parseFloat(state.novec.weight) || 0;
            const shellKg = parseFloat(state.shell.weight) || 0;

            const totalKg = totalBatteryWeightKg + pcsKg + invKg + hvacKg + novKg + shellKg;
            const totalTons = totalKg / 1000;

            document.getElementById('valTotalWeight').innerText = `${totalTons.toFixed(2)} トン`;

            // 3. Axle Load / Distribution Estimate
            // Estimate center of gravity along X axis
            let momentX = 0;
            interactiveObjects.forEach(obj => {
                let w = 100;
                if (obj.userData.id && obj.userData.id.startsWith('rack')) w = packWeightKg * packsPerRack;
                else if (obj.userData.id === 'pcs') w = pcsKg;
                else if (obj.userData.id === 'inv') w = invKg;
                else if (obj.userData.id === 'hvac') w = hvacKg;
                else if (obj.userData.id === 'novec') w = novKg;

                momentX += w * obj.position.x;
            });

            const spec = CONTAINER_SPECS[state.containerSize];
            const cL = spec.length / 1000;
            const frontWeightRatio = 0.5 + (momentX / (cL * totalKg || 1));
            const frontTons = Math.max(0, totalTons * frontWeightRatio);
            const rearTons = Math.max(0, totalTons - frontTons);

            document.getElementById('valAxleLoad').innerText = `F: ${frontTons.toFixed(2)}t / R: ${rearTons.toFixed(2)}t`;
        }
